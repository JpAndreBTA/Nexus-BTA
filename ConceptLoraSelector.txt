        // Selecionar/Deselecionar LoRA
        function toggleLoraSelection(id) {
            if (selectedLoras.hasOwnProperty(id)) {
                delete selectedLoras[id];
            } else {
                selectedLoras[id] = 1.0; // Peso padrão inicial
            }

            renderLoraGrid();
        }

        // Sincroniza e atualiza o painel superior de Sliders modernos de peso (Sem rosa)
        function updateActiveSliders() {
            const container = document.getElementById('activeSelectionsBar');
            const grid = document.getElementById('selectedSlidersGrid');
            const counter = document.getElementById('selectionCounter');

            const activeIds = Object.keys(selectedLoras);
            counter.innerText = activeIds.length;

            if (activeIds.length === 0) {
                container.classList.add('hidden');
                container.classList.remove('flex');
                return;
            }

            container.classList.remove('hidden');
            container.classList.add('flex');

            grid.innerHTML = "";
            activeIds.forEach(id => {
                const item = loraDatabase.find(x => x.id === id);
                if (!item) return;

                const sliderBox = document.createElement('div');
                sliderBox.className = "bg-zinc-950 border border-nexus-border p-2 flex flex-col gap-1.5 select-none";
                sliderBox.innerHTML = `
                    <div class="flex justify-between items-center text-[9px] font-mono">
                        <span class="text-white font-bold truncate pr-2">${item.title}</span>
                        <div class="flex items-center gap-1.5">
                            <span class="text-nexus-red font-bold" id="val-${id}">${selectedLoras[id].toFixed(1)}</span>
                            <button onclick="event.stopPropagation(); toggleLoraSelection('${id}')" class="text-zinc-500 hover:text-nexus-red" title="Remove Lora">
                                <i class="fa-solid fa-circle-xmark"></i>
                            </button>
                        </div>
                    </div>
                    <!-- Slider plano customizado e extremamente limpo (Vermelho NEXUS, sem rosa) -->
                    <input type="range" min="0.1" max="2.0" step="0.1" value="${selectedLoras[id]}" oninput="updateLoraWeight('${id}', this.value)" class="modern-slider w-full">
                `;
                grid.appendChild(sliderBox);
            });
        }

        // Atualizar peso da LoRA de forma instantânea
        function updateLoraWeight(id, value) {
            const parsed = parseFloat(value);
            selectedLoras[id] = parsed;
            
            const text = document.getElementById(`val-${id}`);
            if (text) text.innerText = parsed.toFixed(1);

            // Redesenha apenas os contadores ativos nos cards sem reiniciar o estado de scroll
            const activeCardBadge = document.querySelector(`[onclick*="toggleLoraSelection('${id}')"] .selected-overlay-text`);
            if (activeCardBadge) {
                activeCardBadge.innerText = `Active (${parsed.toFixed(1)})`;
            }
        }

        // Filtro em tempo real digitado na pesquisa
        function filterLoras() {
            renderLoraGrid();
        }

        // Ordenação por seletor dropdown
        function sortLoras() {
            const sortVal = document.getElementById('sortSelector').value;
            
            if (sortVal === "name") {
                loraDatabase.sort((a, b) => a.title.localeCompare(b.title));
            } else if (sortVal === "newest") {
                loraDatabase.sort((a, b) => b.id.localeCompare(a.id));
            } else if (sortVal === "oldest") {
                loraDatabase.sort((a, b) => a.id.localeCompare(b.id));
            } else if (sortVal === "downloads") {
                loraDatabase.sort((a, b) => parseFloat(b.downloads.replace(/K/g, '')) - parseFloat(a.downloads.replace(/K/g, '')));
            }
            renderLoraGrid();
        }

        // Copiar trigger words para a área de transferência
        function copyTrigger(trigger) {
            const el = document.createElement('textarea');
            el.value = trigger;
            document.body.appendChild(el);
            el.select();
            document.execCommand('copy');
            document.body.removeChild(el);

            alert(`Copied trigger: "${trigger}"`);
        }

        // Resetar todos os filtros ativos de busca
        function resetFilters() {
            document.getElementById('loraSearchInput').value = "";
            activeCategory = "all";
            
            const firstButton = document.querySelector('.category-btn');
            if (firstButton) selectCategory('all', firstButton);
            
            selectedLoras = {};
            renderLoraGrid();
        }
